Everything beyond the drug names: the interactions, storage rules, laws, sig codes, and math that show up on the PTCE. Pick a category, flip the card (or press Space), and mark it Got it or Still learning. Progress is saved in this browser.

Keyboard: Space flip · ← → previous / next · K got it · L still learning

fc = import(new URL("../../flashcards.js", document.baseURI).href)
flashcards = fc.flashcards
esc = fc.esc
review = FileAttachment("ptcb_review.csv").csv()
flashcards({
  cards: review,
  storageKey: "pharmtech-review-known",
  idField: "id",
  groupField: "category",
  groupLabel: "Category",
  searchFields: ["question", "answer", "category"],
  placeholder: "e.g. DEA 222, insulin, Clark's rule",
  invalidation,
  modes: {
    qa: {
      label: "Question → Answer",
      front: d => `<div class="fc-big fc-mid">${esc(d.question)}</div>`,
      back:  d => `<div class="fc-row fc-answer">${esc(d.answer)}</div>`
    },
    aq: {
      label: "Answer → Question (harder)",
      front: d => `<div class="fc-kicker">What's the question?</div><div class="fc-row fc-answer">${esc(d.answer)}</div>`,
      back:  d => `<div class="fc-big fc-mid">${esc(d.question)}</div>`
    }
  }
})
